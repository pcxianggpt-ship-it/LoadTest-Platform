# 异步报告生成与项目选择保持 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在生成报告时异步执行 Grafana 自动截图和报告渲染，并让前端各业务模块保持用户最近选择的项目。

**Architecture:** 后端先创建 `pending` 报告并立即返回，再由定时调度串行处理报告任务；报告任务负责按项目 Grafana datasource 配置截图、嵌入图片、更新状态和失败原因。前端通过报告状态轮询展示生成进度，并用一个轻量 composable 将当前项目 ID 持久化到 `localStorage`。

**Tech Stack:** Spring Boot 3.2、MyBatis Plus、SQLite/Flyway、Vue 3、Vue Router、Element Plus、TypeScript。

@Transactional
public void runOnePendingReport() {
    if (hasRunningReport()) {
        return;
    }
    TestReport report = nextPendingReport();
    if (report == null) {
        return;
    }
    markRunning(report);
    try {
        completeReport(report);
    } catch (Exception exception) {
        markFailed(report, exception.getMessage());
    }
}
```

private ResultImageResponse saveRenderedImage(
        TestResult result,
        ResultImageRequest request,
        GrafanaRenderRequest renderRequest,
        GrafanaImageData imageData,
        String imageType,
        String title,
        Integer width,
        Integer height
) {
    String now = OffsetDateTime.now().toString();
    String extension = extension(imageData.contentType());
    Path relativePath = Path.of(String.valueOf(result.getProjectId()), String.valueOf(result.getId()),
            "panel-" + request.getPanelId() + "-" + System.currentTimeMillis() + extension);
    Path absolutePath = rootDir().resolve(relativePath).normalize();

    } catch (IOException exception) {

    }

    TestResultImage image = new TestResultImage();
    image.setResultId(result.getId());
    image.setProjectId(result.getProjectId());
    image.setImageType(firstNonBlank(imageType, "grafana_panel"));

    image.setPanelId(request.getPanelId());
    image.setGrafanaUrl(imageData.renderUrl());
    image.setFilePath(relativePath.toString().replace('\\', '/'));
    image.setContentType(imageData.contentType());
    image.setFileSize((long) imageData.content().length);
    image.setWidth(width);
    image.setHeight(height);
    image.setCreatedAt(now);

@Transactional
public List<ResultImageResponse> exportConfiguredGrafanaImages(Long resultId) {
    TestResult result = getResult(resultId);
    ProjectDatasource datasource = datasource(result.getProjectId());
    GrafanaDatasourceConfig config = parseConfig(datasource.getExtraConfigJson());
    List<GrafanaPanelConfig> panels = config.panels() == null ? List.of() : config.panels();
    if (panels.isEmpty()) {
        throw new IllegalArgumentException("grafana panels are required");
    }
    deleteAutoGeneratedImages(result.getId(), panels);
    return panels.stream()
            .map(panel -> exportConfiguredPanel(result, datasource, config, panel))
            .toList();
}
```

private ResultImageResponse exportConfiguredPanel(
        TestResult result,
        ProjectDatasource datasource,
        GrafanaDatasourceConfig config,
        GrafanaPanelConfig panel
) {
    if (panel.panelId() == null) {
        throw new IllegalArgumentException("grafana panelId is required");
    }
    String dashboardUid = firstNonBlank(panel.dashboardUid(), config.dashboardUid());
    if (dashboardUid == null) {
        throw new IllegalArgumentException("grafana dashboardUid is required");
    }
    Integer width = defaultInt(panel.width(), defaultInt(config.width(), 1200));
    Integer height = defaultInt(panel.height(), defaultInt(config.height(), 700));
    Integer orgId = panel.orgId() == null ? config.orgId() : panel.orgId();
    String dashboardSlug = firstNonBlank(panel.dashboardSlug(), config.dashboardSlug(), "_");
    String theme = firstNonBlank(panel.theme(), config.theme(), "light");
    ResultImageRequest request = new ResultImageRequest();
    request.setDashboardUid(dashboardUid);
    request.setDashboardSlug(dashboardSlug);
    request.setPanelId(panel.panelId());
    request.setOrgId(orgId);
    request.setWidth(width);
    request.setHeight(height);
    request.setTheme(theme);
    request.setTitle(panel.title());
    GrafanaRenderRequest renderRequest = new GrafanaRenderRequest(
            dashboardUid,
            dashboardSlug,
            panel.panelId(),
            toEpochMillis(result.getTimeRangeStart()),
            toEpochMillis(result.getTimeRangeEnd()),
            orgId,
            width,
            height,
            theme
    );
    GrafanaImageData imageData = grafanaImageClient.renderPanel(datasource, renderRequest);
    return saveRenderedImage(result, request, renderRequest, imageData,
            firstNonBlank(panel.imageType(), "grafana_panel"),
            panel.title(),
            width,
            height);
}

- [ ] **Step 1: 新增当前项目 composable**

Create `frontend/src/stores/currentProject.ts`:

```ts
import type { Project } from "../api/projects";

const STORAGE_KEY = "loadtest.currentProjectId";

export function getCurrentProjectId() {
  const rawValue = window.localStorage.getItem(STORAGE_KEY);
  const value = Number(rawValue);
  return Number.isFinite(value) && value > 0 ? value : undefined;
}

export function setCurrentProjectId(projectId?: number) {
  if (!projectId) {
    window.localStorage.removeItem(STORAGE_KEY);
    return;
  }
  window.localStorage.setItem(STORAGE_KEY, String(projectId));
}

export function resolveProjectId(projects: Project[], routeProjectId?: number) {
  const routeId = Number(routeProjectId);
  if (Number.isFinite(routeId) && projects.some((project) => project.id === routeId)) {
    setCurrentProjectId(routeId);
    return routeId;
  }
  const storedId = getCurrentProjectId();
  if (storedId && projects.some((project) => project.id === storedId)) {
    return storedId;
  }
  const fallbackId = projects[0]?.id;
  setCurrentProjectId(fallbackId);
  return fallbackId;
}
```
